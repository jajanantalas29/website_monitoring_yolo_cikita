// ===== BUZZER AKTIF: beep pakai digitalWrite =====
void beep(int t, int d) {
    for (int i = 0; i < t; i++) {
        digitalWrite(buzzerPin, HIGH);
        delay(d);
        digitalWrite(buzzerPin, LOW);
        delay(d);
    }
}

//=====POLA REL KERETA (non-blocking)=====//
// ON-100ms, OFF-100ms, ON-100ms, OFF-600ms, repeat
const int TRAIN_PATTERN[] = {100, 100, 100, 600};
const int TRAIN_STEPS = 4;
int buzzerStep = 0;
unsigned long buzzerLastMs = 0;
bool buzzerActive = false;

void buzzerTrainUpdate() {
    if (!buzzerActive) {
        digitalWrite(buzzerPin, LOW);
        return;
    }
    unsigned long now = millis();
    if (now - buzzerLastMs >= (unsigned long)TRAIN_PATTERN[buzzerStep]) {
        buzzerLastMs = now;
        buzzerStep = (buzzerStep + 1) % TRAIN_STEPS;
        digitalWrite(buzzerPin, (buzzerStep % 2 == 0) ? HIGH : LOW);
    }
}

void buzzerStart() {
    if (!buzzerActive) {
        buzzerActive = true;
        buzzerStep = 0;
        buzzerLastMs = millis();
        digitalWrite(buzzerPin, HIGH);
    }
}

void buzzerStop() {
    buzzerActive = false;
    buzzerStep = 0;
    digitalWrite(buzzerPin, LOW);
}

void handlePing() {
    String msg;
    if (WiFi.status() == WL_CONNECTED) {
        HTTPClient http;
        http.begin(cfg_serverPing);
        int code = http.GET();
        if (code > 0) {
            serverConnected = true;
            msg = "<div class='alert alert-ok'>\xE2\x9C\x93 Server merespons HTTP " + String(code) + "</div>";
        } else {
            serverConnected = false;
            msg = "<div class='alert alert-err'>\xE2\x9C\x97 Gagal terhubung ke server</div>";
        }
        http.end();
    } else {
        msg = "<div class='alert alert-err'>\xE2\x9C\x97 WiFi tidak terhubung</div>";
    }
    msg += "<div class='actions'><a href='/server' class='btn btn-primary btn-sm'>← Kembali</a></div>";
    server.send(200, "text/html", buildPage(msg, "server"));
}

void handleOtaUpload() {
    HTTPUpload& u = server.upload();
    if (u.status == UPLOAD_FILE_START) {
        if (!Update.begin(UPDATE_SIZE_UNKNOWN)) Update.printError(Serial);
    } else if (u.status == UPLOAD_FILE_WRITE) {
        if (Update.write(u.buf, u.currentSize) != u.currentSize) Update.printError(Serial);
    } else if (u.status == UPLOAD_FILE_END) {
        if (Update.end(true)) {
            server.send(200, "text/plain", "OK");
            delay(1000);
            ESP.restart();
        } else {
            server.send(500, "text/plain", "OTA FAILED");
            Update.printError(Serial);
        }
    }
}

void setupOTA() {
    ArduinoOTA.setHostname("museum-esp32");
    ArduinoOTA.setPassword(cfg_otaPassword.c_str());
    ArduinoOTA.onStart([]() { digitalWrite(ledPin, HIGH); });
    ArduinoOTA.onEnd([]() { beep(3, 100); });
    ArduinoOTA.begin();
}

void setup() {
    Serial.begin(9600);
    pinMode(trigPin, OUTPUT);
    pinMode(echoPin, INPUT);
    pinMode(ledPin, OUTPUT);
    pinMode(buzzerPin, OUTPUT);
    
    digitalWrite(buzzerPin, LOW);
    
    loadConfig();
    WiFi.config(parseIP(cfg_staticIP), parseIP(cfg_gateway), parseIP(cfg_subnet));
    WiFi.begin(cfg_ssid.c_str(), cfg_password.c_str());
    
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println("\nConnected: " + WiFi.localIP().toString());
    
    setupOTA();
    
    server.on("/", HTTP_GET, handleRoot);
    server.on("/server", HTTP_GET, handleServer);
    server.on("/server", HTTP_POST, handleServer);
    server.on("/sensor", HTTP_GET, handleSensor);
    server.on("/sensor", HTTP_POST, handleSensor);
    server.on("/wifi", HTTP_GET, handleWifi);
    server.on("/wifi", HTTP_POST, handleWifi);
    server.on("/ping", HTTP_GET, handlePing);
    server.on("/restart", HTTP_GET, handleRestart);
    server.on("/ota", HTTP_GET, handleOtaPage);
    server.on("/ota/upload", HTTP_POST, [](){}, handleOtaUpload);
    server.begin();
    
    HTTPClient http;
    http.begin(cfg_serverPing);
    if (http.GET() > 0) {
        serverConnected = true;
        beep(2, 200);
    }
    http.end();
}

void loop() {
    ArduinoOTA.handle();
    server.handleClient();
    
    digitalWrite(trigPin, LOW);
    delayMicroseconds(2);
    digitalWrite(trigPin, HIGH);
    delayMicroseconds(10);
    digitalWrite(trigPin, LOW);
    
    // Add timeout so it doesn't block forever if no echo
    duration = pulseIn(echoPin, HIGH, 30000); 
    
    if (duration == 0) {
        distance_cm = 999; // out of range
    } else {
        distance_cm = duration * 0.034 / 2;
    }
    
    Serial.println(distance_cm);
    
    if (distance_cm > 0 && distance_cm <= cfg_violationDist) {
        digitalWrite(ledPin, HIGH);
        buzzerStart();
        
        if (serverConnected && millis() - lastViolationTime >= (unsigned long)cfg_violationDelay) {
            HTTPClient http;
            http.begin(cfg_serverTrigger);
            int code = http.GET();
            Serial.println(code > 0 ? "TRIGGER OK" : "TRIGGER FAIL");
            http.end();
            lastViolationTime = millis();
        }
    } else {
        digitalWrite(ledPin, LOW);
        buzzerStop();
    }
    
    buzzerTrainUpdate();
    delay(cfg_loopDelay);
}
